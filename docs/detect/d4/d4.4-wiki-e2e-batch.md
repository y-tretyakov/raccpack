## 2. E2E

```bash
ROOT=/tmp/racc-batch-root
DEN=/tmp/racc-batch-den
rm -rf "$ROOT" "$DEN"
# two mini projects with .env + marker
RACCPACK_PASSPHRASE=test \
  cargo run -p raccpack-cli -- raid --root "$ROOT" --den "$DEN" --yes
test "$(find "$DEN/secrets" -name '*.age' | wc -l)" -ge 2
test "$(find "$DEN/packs" -name '*.tar.zst' | wc -l)" -ge 2
```
